//直接插入排序
void directsort(int num[],int n)
{
    for(int i = 2; i <= n ; i ++)
    {
        num[0] = num[i]
        for(int j = i - 1; num[j] > num[0] ; j ++)
        {
            num[j + 1] = num[j];
        }
        num[j + 1] = num[0];
    }
}

//折半插入排序
void zhebansort(int num[],int n)
{
    int low,high,mid;
    for(int i = 2 ; i <= n ; i++)
    {
        num[0] = num[i];
        low = 1;
        high = i - 1;
        while(low <= high)
        {
            mid = (low + high) / 2;
            if(num[0] < num[mid]) high = mid -1;
            else
            {
                low = mid + 1;
            }
        }
        for(int j = i - 1; j >= high + 1; j --)
        {
            num[j + 1] = num[j];
        }
        num[high + 1] = num[0];
    }
}

void bubblesort(int num[],int n)
{
    int i,j,temp;
    for(i = 1 ; i < n ; i ++)
    {
        int exchange = 0;
        for(j = 1 ; j < n - i; j ++)
        {
            if(num[j] > num[j + 1])
            {
                num[j + 1] = temp;
                num[j + 1] = num[j];
                num[j] = num[j + 1];
                exchange = 1;
            }
        }
        if(!exchange) break;
    }
}
